# Host-built tool executables used to generate the dictionaries when the
# host cannot execute the just-built target tools (cross compiling without
# emulation). Dictionaries are architecture-independent data, so generating
# them with host tools produces the same result as generating them on the
# target. Both tools must be built from the same OpenCC source tree as this
# build: dictionaries are generated by opencc_dict and the ST phrase
# generator executes the opencc CLI, and other versions may produce
# inconsistent dictionaries or fail halfway.
set(OPENCC_DICT_EXECUTABLE "" CACHE FILEPATH "Path to a host-built opencc_dict executable used to generate dictionaries when cross compiling; must be built from the same OpenCC source version as this build")
set(OPENCC_CLI_EXECUTABLE "" CACHE FILEPATH "Path to a host-built opencc CLI executable used to generate dictionaries when cross compiling; must be built from the same OpenCC source version as this build")

# Building the dictionaries executes the just-built target tools (opencc_dict
# and the opencc CLI) at build time. When the host cannot execute them and no
# host tools are provided, default to skipping the data build instead of
# failing mid-build. The default can be overridden explicitly, e.g. when the
# host can emulate the target (Rosetta 2).
include("${CMAKE_CURRENT_SOURCE_DIR}/cmake/OpenCCRunnability.cmake")

if(OPENCC_DICT_EXECUTABLE OR OPENCC_CLI_EXECUTABLE)
  if(NOT OPENCC_DICT_EXECUTABLE OR NOT OPENCC_CLI_EXECUTABLE)
    message(FATAL_ERROR
      "OPENCC_DICT_EXECUTABLE and OPENCC_CLI_EXECUTABLE must be provided "
      "together: building the dictionaries runs both the opencc_dict tool "
      "and the opencc CLI.")
  endif()
  foreach(_opencc_host_tool_var OPENCC_DICT_EXECUTABLE OPENCC_CLI_EXECUTABLE)
    if(NOT EXISTS "${${_opencc_host_tool_var}}")
      message(FATAL_ERROR
        "${_opencc_host_tool_var} does not exist: ${${_opencc_host_tool_var}}")
    endif()
  endforeach()
  # The host tools must at least be runnable on this host (existence above
  # says nothing about their dynamic libraries); probe opencc_dict --help
  # like the standalone Jieba plugin build does for its installed tool.
  opencc_host_tool_command(_opencc_host_dict_command "${OPENCC_DICT_EXECUTABLE}")
  execute_process(
    COMMAND ${_opencc_host_dict_command} --help
    RESULT_VARIABLE _opencc_host_dict_help_result
    OUTPUT_VARIABLE _opencc_host_dict_help
    ERROR_VARIABLE _opencc_host_dict_help_err
  )
  if(NOT _opencc_host_dict_help_result EQUAL 0)
    message(FATAL_ERROR
      "Failed to run host tool ${OPENCC_DICT_EXECUTABLE} (exit status "
      "${_opencc_host_dict_help_result}): ${_opencc_host_dict_help_err}")
  endif()
  set(OPENCC_USING_HOST_TOOLS ON)
else()
  set(OPENCC_USING_HOST_TOOLS OFF)
endif()

# Whether OPENCC_BUILD_DATA was provided explicitly — a plain variable
# already defined (a parent project setting it before add_subdirectory(),
# which CMP0077 option() semantics would honor) or a cache entry (-D or a
# previous configure) — so that the warning only fires when the default was
# flipped automatically rather than chosen by the user. When not provided,
# the value stays a plain variable recomputed on every configure: caching it
# would make the first automatic OFF look user-provided on later configures
# (e.g. after host tools are added) and silently ignore them.
get_property(_OPENCC_BUILD_DATA_CACHED CACHE OPENCC_BUILD_DATA PROPERTY TYPE SET)
if(DEFINED OPENCC_BUILD_DATA OR _OPENCC_BUILD_DATA_CACHED)
  set(_OPENCC_BUILD_DATA_PROVIDED ON)
else()
  set(_OPENCC_BUILD_DATA_PROVIDED OFF)
endif()
set(_OPENCC_BUILD_DATA_AUTO_OFF OFF)
if(NOT _OPENCC_BUILD_DATA_PROVIDED)
  if(NOT OPENCC_CAN_RUN_TARGET_TOOLS
      AND NOT OPENCC_DICT_EXECUTABLE AND NOT OPENCC_CLI_EXECUTABLE)
    set(OPENCC_BUILD_DATA OFF)
    set(_OPENCC_BUILD_DATA_AUTO_OFF ON)
  else()
    set(OPENCC_BUILD_DATA ON)
  endif()
endif()
if(_OPENCC_BUILD_DATA_AUTO_OFF)
  message(WARNING
    "Cross compiling for an architecture the host cannot execute: "
    "OPENCC_BUILD_DATA defaulted to OFF, so no dictionaries or config "
    "files will be built or installed. Provide host-built tools via "
    "OPENCC_DICT_EXECUTABLE and OPENCC_CLI_EXECUTABLE to build them "
    "anyway, or set OPENCC_BUILD_DATA=ON explicitly if the host can run "
    "the target binaries (e.g. through Rosetta 2 or "
    "CMAKE_CROSSCOMPILING_EMULATOR).")
endif()
if(OPENCC_USING_HOST_TOOLS AND _OPENCC_BUILD_DATA_PROVIDED AND NOT OPENCC_BUILD_DATA)
  message(WARNING
    "Host tools are provided (OPENCC_DICT_EXECUTABLE and "
    "OPENCC_CLI_EXECUTABLE) but OPENCC_BUILD_DATA is explicitly OFF: they "
    "will not be used and no dictionaries or config files will be built.")
endif()
if(OPENCC_BUILD_DATA AND NOT OPENCC_BUILD_TOOLS AND NOT OPENCC_USING_HOST_TOOLS)
  message(FATAL_ERROR
    "OPENCC_BUILD_DATA requires OPENCC_BUILD_TOOLS, or host tools via "
    "OPENCC_DICT_EXECUTABLE and OPENCC_CLI_EXECUTABLE.")
endif()

######## Directory
# Installation directories follow GNU Coding Standards via CMake's GNUInstallDirs module:
# https://cmake.org/cmake/help/latest/module/GNUInstallDirs.html
#
# Default values and parity with previously hardcoded paths
# (shown for CMAKE_INSTALL_PREFIX=/usr/local and /usr):
#
#   CMAKE_INSTALL_INCLUDEDIR     "include"  -> /usr/local/include  or  /usr/include
#     (replaces: ${CMAKE_INSTALL_PREFIX}/include)
#
#   CMAKE_INSTALL_SYSCONFDIR     "etc"      -> /usr/local/etc      or  /etc
#     (replaces: ${CMAKE_INSTALL_PREFIX}/etc)
#     NOTE: when prefix is /usr the full path is /etc per GNU Coding Standards,
#           whereas the old code would have produced /usr/etc.
#
#   CMAKE_INSTALL_LIBDIR         "lib"      -> /usr/local/lib      or  /usr/lib
#     (also resolves to lib64 or lib/<multiarch> on applicable platforms)
#     (replaces: ${CMAKE_INSTALL_PREFIX}/lib[64])
#
#   CMAKE_INSTALL_DATAROOTDIR    "share"    -> /usr/local/share    or  /usr/share
#     (replaces: ${CMAKE_INSTALL_PREFIX}/share)

# All DIR_* variables use relative paths so that install() destinations and
# CMake package config files remain relocatable.
set (DIR_INCLUDE ${CMAKE_INSTALL_INCLUDEDIR})
set (DIR_ETC ${CMAKE_INSTALL_SYSCONFDIR})
set (DIR_LIBRARY ${CMAKE_INSTALL_LIBDIR})
set (DIR_SHARE ${CMAKE_INSTALL_DATAROOTDIR})

if (DEFINED SHARE_INSTALL_PREFIX)
  set (DIR_SHARE ${SHARE_INSTALL_PREFIX})
endif (DEFINED SHARE_INSTALL_PREFIX)

if (DEFINED INCLUDE_INSTALL_DIR)
  set (DIR_INCLUDE ${INCLUDE_INSTALL_DIR})
endif (DEFINED INCLUDE_INSTALL_DIR)

if (DEFINED SYSCONF_INSTALL_DIR)
  set (DIR_ETC ${SYSCONF_INSTALL_DIR})
endif (DEFINED SYSCONF_INSTALL_DIR)

if (DEFINED LIB_INSTALL_DIR)
  set (DIR_LIBRARY ${LIB_INSTALL_DIR})
endif (DEFINED LIB_INSTALL_DIR)

set (DIR_SHARE_OPENCC ${DIR_SHARE}/opencc)
if (WIN32)
  set (DIR_PLUGIN bin/plugins)
else ()
  set (DIR_PLUGIN ${DIR_LIBRARY}/opencc/plugins)
endif ()

# Compute absolute paths needed for compile-time macros (PKGDATADIR/LOCALEDIR)
# and for the pkg-config file.  We must handle both the GNUInstallDirs relative
# defaults and user-supplied absolute override paths (SHARE_INSTALL_PREFIX etc.).
foreach (_gnu_dir SHARE INCLUDE LIBRARY)
  if (IS_ABSOLUTE "${DIR_${_gnu_dir}}")
    set (DIR_${_gnu_dir}_FULL "${DIR_${_gnu_dir}}")
  else ()
    set (DIR_${_gnu_dir}_FULL "${CMAKE_INSTALL_PREFIX}/${DIR_${_gnu_dir}}")
  endif ()
endforeach ()

set (DIR_SHARE_OPENCC_FULL ${DIR_SHARE_FULL}/opencc)
set (DIR_SHARE_LOCALE_FULL ${DIR_SHARE_FULL}/locale)
if (IS_ABSOLUTE "${DIR_PLUGIN}")
  set (DIR_PLUGIN_FULL "${DIR_PLUGIN}")
else ()
  set (DIR_PLUGIN_FULL "${CMAKE_INSTALL_PREFIX}/${DIR_PLUGIN}")
endif ()
